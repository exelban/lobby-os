One Docker container. A home for all your services.

docker run -d --name lobby-os \
  -p 8080:8080 \
  -v lobby_os_data:/srv/data \
  exelban/lobby-os:latest
services:
  lobby-os:
    image: exelban/lobby-os:latest
    ports:
      - "8080:8080"
    volumes:
      - lobby_os_data:/srv/data

volumes:
  lobby_os_data:

Open localhost:8080 to get started.
